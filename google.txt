const fileToBase64 = (file) => new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = () => resolve(reader.result);
    reader.onerror = error => reject(error);
});

const getImageDimensions = (src) => new Promise((resolve) => {
    const img = new Image();
    img.onload = () => resolve({ w: img.width, h: img.height });
    img.onerror = () => resolve({ w: 300, h: 300 }); 
    img.src = src;
});

const calculateNodeSize = (w, h, maxW = 360) => {
    if (w <= maxW) return { w, h };
    const ratio = maxW / w;
    return { w: maxW, h: Math.round(h * ratio) };
};

function useDebounceCallback(callback, delay) {
    const timeoutRef = useRef(null);
    return useCallback((...args) => {
        if (timeoutRef.current) clearTimeout(timeoutRef.current);
        timeoutRef.current = setTimeout(() => {
            callback(...args);
        }, delay);
    }, [callback, delay]);
}

                body: JSON.stringify(payload)
            });
            if (!submitRes.ok) throw new Error(`Volcengine API HTTP Error: ${submitRes.status}`);
            const submitData = await submitRes.json();
            if (!submitData.id) throw new Error("API 未返回 Task ID");
            const taskId = submitData.id;

            // 轮询状态
            let attempts = 0;
            while (attempts < 60) {

    } catch (error) {

        await new Promise(r => setTimeout(r, 4000));

  }

// --- 主路由 ---
export default function AppRoot() {
    const [user, setUser] = useState(null);
    const [authLoading, setAuthLoading] = useState(true);
    const [currentProjectId, setCurrentProjectId] = useState(null);

    useEffect(() => {
        const initAuth = async () => {
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) await signInWithCustomToken(auth, __initial_auth_token);
                else await signInAnonymously(auth);
            } catch (err) { console.error("Auth error", err); }
        };
        initAuth();
        const unsubscribe = onAuthStateChanged(auth, (u) => {
            setUser(u); setAuthLoading(false);
            const urlParams = new URLSearchParams(window.location.search);
            const sharedProjectId = urlParams.get('projectId');
            if (sharedProjectId) setCurrentProjectId(sharedProjectId);
        });
        return () => unsubscribe();
    }, []);

    if (authLoading) return <div className="w-full h-screen flex items-center justify-center bg-neutral-50"><Loader2 className="animate-spin text-blue-500" size={32} /></div>;
    if (!user) return <div className="w-full h-screen flex items-center justify-center">认证失败，请刷新页面重试。</div>;

    return (
        <div className="w-full h-screen font-sans text-neutral-800 bg-neutral-50">
            {currentProjectId ? (
                <TapnowCanvas user={user} projectId={currentProjectId} onBack={() => { setCurrentProjectId(null); window.history.pushState({}, '', window.location.pathname); }} />
            ) : <Dashboard user={user} onOpenProject={setCurrentProjectId} />}
        </div>
    );
}

// --- 项目控制台 ---
function Dashboard({ user, onOpenProject }) {
    const [projects, setProjects] = useState([]);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        if (!user) return;
        const projectsRef = collection(db, 'artifacts', appId, 'public', 'data', 'tapnow_projects');
        const unsubscribe = onSnapshot(projectsRef, (snapshot) => {
            const allProjects = [];
            snapshot.forEach(doc => {
                const data = doc.data();
                if (data.ownerId === user.uid) allProjects.push({ id: doc.id, ...data });
            });
            allProjects.sort((a, b) => b.updatedAt - a.updatedAt);
            setProjects(allProjects);
            setLoading(false);
        });
        return () => unsubscribe();
    }, [user]);

    const createNewProject = async () => {
        const projectId = generateId();
        const newProject = {
            id: projectId, name: `未命名画布 ${new Date().toLocaleDateString()}`, ownerId: user.uid,
            viewport: { x: 0, y: 0, zoom: 1 }, nodes: [], edges: [], createdAt: Date.now(), updatedAt: Date.now()
        };
        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'tapnow_projects', projectId), newProject);
        onOpenProject(projectId);
    };

    const deleteProject = async (e, id) => {
        e.stopPropagation();
        if (window.confirm("确定要删除这个项目吗？")) await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'tapnow_projects', id));
    };

    return (
        <div className="max-w-5xl mx-auto p-8 h-full overflow-y-auto">
            <div className="flex items-center justify-between mb-8">
                <div>
                    <h1 className="text-3xl font-bold text-neutral-900 flex items-center gap-3"><FolderKanban className="text-blue-500" /> Tapnow 云端工作区</h1>
                    <p className="text-neutral-500 mt-2">管理你的所有无限画布项目。数据已云端安全同步。</p>
                </div>
                <div className="flex items-center gap-4 text-sm text-neutral-500 bg-white px-4 py-2 rounded-full shadow-sm border border-neutral-200">
                    <span className="w-2 h-2 rounded-full bg-green-500"></span>在线状态: {user.uid.substring(0, 6)}...
                </div>
            </div>
            <div className="grid grid-cols-1 md:grid-cols-3 lg:grid-cols-4 gap-6">
                <div onClick={createNewProject} className="bg-white border-2 border-dashed border-neutral-300 rounded-2xl h-48 flex flex-col items-center justify-center cursor-pointer hover:border-blue-500 hover:bg-blue-50 transition-all group">
                    <div className="w-12 h-12 rounded-full bg-blue-100 text-blue-500 flex items-center justify-center mb-3 group-hover:scale-110 transition-transform"><Plus size={24} /></div>
                    <span className="font-medium text-neutral-600 group-hover:text-blue-600">新建画布项目</span>
                </div>
                {loading ? <div className="col-span-3 flex justify-center py-10"><Loader2 className="animate-spin text-neutral-400" /></div> : projects.map(proj => (
                    <div key={proj.id} onClick={() => onOpenProject(proj.id)} className="bg-white border border-neutral-200 rounded-2xl h-48 p-5 flex flex-col relative cursor-pointer hover:shadow-lg hover:border-blue-300 transition-all group">
                        <div className="flex-1">
                            <h3 className="font-bold text-lg text-neutral-800 line-clamp-1">{proj.name}</h3>
                            <p className="text-xs text-neutral-400 mt-2">包含 {proj.nodes?.length || 0} 个节点</p>
                        </div>
                        <div className="flex items-center justify-between mt-4 pt-4 border-t border-neutral-100">
                            <span className="text-xs text-neutral-400">{new Date(proj.updatedAt).toLocaleDateString()}</span>
                            <button onClick={(e) => deleteProject(e, proj.id)} className="text-neutral-300 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity"><Trash2 size={16} /></button>
                        </div>
                    </div>
                ))}
            </div>
        </div>
    );
}

// --- 核心画布组件 ---
function TapnowCanvas({ user, projectId, onBack }) {
  const [docState, setDocState] = useState({ viewport: { x: 0, y: 0, zoom: 1 }, nodes: [], edges: [], name: '加载中...' });
  const docRefState = useRef({ updatedAt: 0 }); 
  const viewportRef = useRef(docState.viewport); 
  const historyRef = useRef({ past: [], future: [] });

  useEffect(() => { viewportRef.current = docState.viewport; }, [docState.viewport]);

  const [selection, setSelection] = useState([]);
  const [dragState, setDragState] = useState(null); 
  const [selectionBox, setSelectionBox] = useState(null);
  const [contextMenu, setContextMenu] = useState(null); 
  const [isSaving, setIsSaving] = useState(false);
  const [drawingNodeId, setDrawingNodeId] = useState(null); 
  const [visiblePrompts, setVisiblePrompts] = useState({}); 
  
  const [edgeMenu, setEdgeMenu] = useState(null);

  const containerRef = useRef(null);
  
  useEffect(() => {
    if (!user || !projectId) return;
    const projectRef = doc(db, 'artifacts', appId, 'public', 'data', 'tapnow_projects', projectId);
    const unsubscribe = onSnapshot(projectRef, (snapshot) => {
        if (snapshot.exists()) {
            const remoteData = snapshot.data();
            if (remoteData.updatedAt > docRefState.current.updatedAt || docRefState.current.updatedAt === 0) {
                setDocState(remoteData);
                docRefState.current.updatedAt = remoteData.updatedAt;
            }
        }
    }, (err) => console.error("Sync error:", err));
    return () => unsubscribe();
  }, [user, projectId]);

  const saveToCloud = useDebounceCallback(async (newData) => {
    if (!user) return;
    setIsSaving(true);
    const updatedData = { ...newData, updatedAt: Date.now() };
    docRefState.current.updatedAt = updatedData.updatedAt; 
    try { await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'tapnow_projects', projectId), updatedData, { merge: true }); } catch (err) { }
    setTimeout(() => setIsSaving(false), 500);
  }, 800);

  const updateDocLocal = (updater) => {
      setDocState(prev => {
          const next = typeof updater === 'function' ? updater(prev) : { ...prev, ...updater };
          saveToCloud(next);
          return next;
      });
  };

  const pushHistory = (stateSnapshot) => {
      historyRef.current.past.push({ nodes: stateSnapshot.nodes, edges: stateSnapshot.edges });
      if (historyRef.current.past.length > 30) historyRef.current.past.shift();
      historyRef.current.future = [];
  };

  useEffect(() => {
      if (docState.nodes.length === 0) return;
      const pendingNodes = docState.nodes.filter(n => n._isLoading && n._taskParams);
      if (pendingNodes.length > 0) pendingNodes.forEach(node => resumeTask(node));
  }, [docState.nodes.length]); 

  const resumeTask = async (targetNode) => {
      const params = targetNode._taskParams;
      try {
          let result;
          if (targetNode.type === 'video') {
             result = await ApiClient.generateVideo(params);
             updateNodeStrict(targetNode.id, { type: 'video_media', src: result.url, _isLoading: false, _taskParams: null, _error: null });
          } else {
             result = await ApiClient.generateImage(params);
             updateNodeStrict(targetNode.id, { type: 'picture', src: result.urls[0], _isLoading: false, _taskParams: null, _error: null });
          }
      } catch (err) {
          updateNodeStrict(targetNode.id, { _isLoading: false, _error: "执行失败: " + err.message, _taskParams: null });
      }
  };

  const updateNodeStrict = async (nodeId, data) => {
      let nextDocSnapshot;
      setDocState(prev => {
          const nextNodes = prev.nodes.map(n => n.id === nodeId ? { ...n, ...data } : n);
          nextDocSnapshot = { ...prev, nodes: nextNodes, updatedAt: Date.now() };
          docRefState.current.updatedAt = nextDocSnapshot.updatedAt;
          return nextDocSnapshot;
      });
      if (nextDocSnapshot) {
          try { await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'tapnow_projects', projectId), nextDocSnapshot, { merge: true }); } catch (err) { }
      }
  };

  const screenToCanvas = (sx, sy) => ({ x: (sx - docState.viewport.x) / docState.viewport.zoom, y: (sy - docState.viewport.y) / docState.viewport.zoom });
  
  const handleWheel = (e) => {
    if (e.ctrlKey || e.metaKey) { 
      e.preventDefault();
      const zoomSensitivity = 0.001; const delta = -e.deltaY * zoomSensitivity;
      const newZoom = Math.min(Math.max(0.1, docState.viewport.zoom * Math.exp(delta)), 5);
      const rect = containerRef.current.getBoundingClientRect();
      const mouseX = e.clientX - rect.left; const mouseY = e.clientY - rect.top;
      const newX = mouseX - (mouseX - docState.viewport.x) * (newZoom / docState.viewport.zoom);
      const newY = mouseY - (mouseY - docState.viewport.y) * (newZoom / docState.viewport.zoom);
      updateDocLocal({ viewport: { x: newX, y: newY, zoom: newZoom } });
    } else {
      updateDocLocal(d => ({ ...d, viewport: { ...d.viewport, x: d.viewport.x - e.deltaX, y: d.viewport.y - e.deltaY } }));
    }
  };

  const handlePointerDown = (e) => {
    if (e.button !== 0) return;
    setContextMenu(null); setDrawingNodeId(null); setEdgeMenu(null);
    const isNodeHit = e.target.closest('[data-node-id]'); const isEdgeHit = e.target.closest('[data-edge-id]');
    
    if (!isNodeHit && !isEdgeHit) {
      const startPos = { x: e.clientX, y: e.clientY }; const startVp = { ...docState.viewport };
      if (e.shiftKey) { setDragState({ type: 'box', start: startPos }); setSelection([]); } 
      else { setDragState({ type: 'pan', start: startPos, startVp }); setSelection([]); }
    }
  };

  const handleDoubleClick = (e) => {
    if (e.target.closest('[data-node-id]')) return;
    const rect = containerRef.current.getBoundingClientRect();
    const cx = screenToCanvas(e.clientX - rect.left, e.clientY - rect.top).x;
    const cy = screenToCanvas(e.clientX - rect.left, e.clientY - rect.top).y;
    setContextMenu({ x: e.clientX, y: e.clientY, cx, cy });
  };

  useEffect(() => {
    const handlePointerMove = (e) => {
      if (!dragState) return;
      if (dragState.type === 'pan') {
        const dx = e.clientX - dragState.start.x; const dy = e.clientY - dragState.start.y;
        setDocState(d => ({ ...d, viewport: { ...d.viewport, x: dragState.startVp.x + dx, y: dragState.startVp.y + dy } }));
      } else if (dragState.type === 'box') {
        setSelectionBox({ x: Math.min(dragState.start.x, e.clientX), y: Math.min(dragState.start.y, e.clientY), w: Math.abs(dragState.start.x - e.clientX), h: Math.abs(dragState.start.y - e.clientY) });
      } else if (dragState.type === 'node') {
        const dx = (e.clientX - dragState.start.x) / docState.viewport.zoom; const dy = (e.clientY - dragState.start.y) / docState.viewport.zoom;
        setDocState(d => {
          const newNodes = d.nodes.map(n => dragState.startNodePos[n.id] ? { ...n, position: { x: dragState.startNodePos[n.id].x + dx, y: dragState.startNodePos[n.id].y + dy } } : n);
          return { ...d, nodes: newNodes };
        });
      } else if (dragState.type === 'edge') {
        setDragState(prev => ({ ...prev, currentPos: screenToCanvas(e.clientX - containerRef.current.getBoundingClientRect().left, e.clientY - containerRef.current.getBoundingClientRect().top) }));
      }
    };

    const handlePointerUp = (e) => {
      if (dragState?.type === 'pan' || dragState?.type === 'node') {
          saveToCloud(docState);
      } else if (dragState?.type === 'box' && selectionBox) {
        const rect = containerRef.current.getBoundingClientRect();
        const boxWorld = {
          x1: screenToCanvas(selectionBox.x - rect.left, 0).x, y1: screenToCanvas(0, selectionBox.y - rect.top).y,
          x2: screenToCanvas(selectionBox.x + selectionBox.w - rect.left, 0).x, y2: screenToCanvas(0, selectionBox.y + selectionBox.h - rect.top).y,
        };
        const selectedIds = docState.nodes.filter(n => n.position.x < boxWorld.x2 && (n.position.x + n.size.w) > boxWorld.x1 && n.position.y < boxWorld.y2 && (n.position.y + n.size.h) > boxWorld.y1).map(n => n.id);
        setSelection(selectedIds);
      } else if (dragState?.type === 'edge') {
        const targetNodeEl = e.target.closest('[data-node-id]');
        if (targetNodeEl) {
           const targetId = targetNodeEl.getAttribute('data-node-id');
           if (targetId !== dragState.nodeId) {
               const fromId = dragState.anchor === 'right' ? dragState.nodeId : targetId;
               const toId = dragState.anchor === 'right' ? targetId : dragState.nodeId;
               pushHistory(docState);
               updateDocLocal(d => ({ ...d, edges: [...d.edges, { id: generateId(), from: fromId, to: toId, label: 'ref' }] }));
           }
        } else {
           setEdgeMenu({
               fromNodeId: dragState.nodeId,
               anchor: dragState.anchor,
               x: e.clientX, y: e.clientY,
               cx: dragState.currentPos.x, cy: dragState.currentPos.y
           });
        }
      }
      setDragState(null); setSelectionBox(null);
    };

    window.addEventListener('pointermove', handlePointerMove); window.addEventListener('pointerup', handlePointerUp);
    return () => { window.removeEventListener('pointermove', handlePointerMove); window.removeEventListener('pointerup', handlePointerUp); };
  }, [dragState, docState.viewport, selectionBox, selection, docState]);

  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;
      
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') {
          e.preventDefault();
          if (e.shiftKey) { 
              if (historyRef.current.future.length > 0) {
                  const nextState = historyRef.current.future.pop();
                  setDocState(prev => {
                      historyRef.current.past.push({ nodes: prev.nodes, edges: prev.edges });
                      const nextDoc = { ...prev, nodes: nextState.nodes, edges: nextState.edges, updatedAt: Date.now() };
                      saveToCloud(nextDoc); return nextDoc;
                  });
              }
          } else { 
              if (historyRef.current.past.length > 0) {
                  const prevState = historyRef.current.past.pop();
                  setDocState(prev => {
                      historyRef.current.future.push({ nodes: prev.nodes, edges: prev.edges });
                      const nextDoc = { ...prev, nodes: prevState.nodes, edges: prevState.edges, updatedAt: Date.now() };
                      saveToCloud(nextDoc); return nextDoc;
                  });
              }
          }
          return;
      }

      if (e.key === 'Delete' || e.key === 'Backspace') {
        pushHistory(docState);
        updateDocLocal(d => ({
          ...d, nodes: d.nodes.filter(n => !selection.includes(n.id)),
          edges: d.edges.filter(e => !selection.includes(e.id) && !selection.includes(e.from) && !selection.includes(e.to))
        }));
        setSelection([]);
      }
      if (e.key === 'Escape') { setSelection([]); setDrawingNodeId(null); setEdgeMenu(null); }
    };

    const handlePaste = async (e) => {
      if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;
      const items = e.clipboardData?.items; if (!items) return;
      const centerX = (window.innerWidth / 2 - viewportRef.current.x) / viewportRef.current.zoom;
      const centerY = (window.innerHeight / 2 - viewportRef.current.y) / viewportRef.current.zoom;
      
      const files = Array.from(items).filter(i => i.type.startsWith('image/') || i.type.startsWith('video/')).map(i => i.getAsFile());
      if (files.length > 0) pushHistory(docState);
      
      for (let i = 0; i < files.length; i++) {
          const file = files[i];
          try {
              if (file.type.startsWith('image/')) {
                  const base64 = await fileToBase64(file);
                  const dims = await getImageDimensions(base64); 
                  const size = calculateNodeSize(dims.w, dims.h);
                  createNode('picture', { x: centerX + i * 30, y: centerY + i * 30 }, { src: base64, size });
              } else if (file.type.startsWith('video/')) {
                  const url = URL.createObjectURL(file);
                  createNode('video_media', { x: centerX + i * 30, y: centerY + i * 30 }, { src: url, size: { w: 360, h: 200 } });
              }
          } catch (err) { console.error("粘贴文件失败", err); }
      }
    };

    window.addEventListener('keydown', handleKeyDown); window.addEventListener('paste', handlePaste);
    return () => { window.removeEventListener('keydown', handleKeyDown); window.removeEventListener('paste', handlePaste); };
  }, [selection, docState]);

  const createNode = (type, position, initialData = {}) => {
    pushHistory(docState);
    let defaultSize = { w: 300, h: 300 };
    if (type === 'text') defaultSize = { w: 260, h: 100 };
    if (type === 'video' || type === 'video_media') defaultSize = { w: 360, h: 200 };
    
    const size = initialData.size || defaultSize;
    const newNode = {
      id: generateId(), type, position, size,
      content: type === 'text' ? '' : undefined, src: type !== 'text' ? '' : undefined, ...initialData
    };
    updateDocLocal(d => ({ ...d, nodes: [...d.nodes, newNode] }));
    setSelection([newNode.id]);
  };

  const handleNodePointerDown = (e, id) => {
    e.stopPropagation();
    let newSelection = [...selection];
    if (!e.shiftKey && !selection.includes(id)) newSelection = [id];
    else if (e.shiftKey) newSelection = selection.includes(id) ? selection.filter(i => i !== id) : [...selection, id];
    setSelection(newSelection);

    pushHistory(docState);

    const startNodePos = {};
    docState.nodes.forEach(n => { if (newSelection.includes(n.id)) startNodePos[n.id] = { ...n.position }; });
    if (drawingNodeId !== id) setDragState({ type: 'node', start: { x: e.clientX, y: e.clientY }, startNodePos });
  };

  const handleCreateFromEdge = (type, menuData) => {
      pushHistory(docState);
      const newNodeId = generateId();
      let size = { w: 300, h: 300 };
      if (type === 'video') size = { w: 360, h: 200 };
      if (type === 'text') size = { w: 260, h: 100 };
      
      const newNode = {
          id: newNodeId, type, position: { x: menuData.cx, y: menuData.cy }, size,
          content: type === 'text' ? '' : undefined, src: undefined
      };
      
      const fromId = menuData.anchor === 'right' ? menuData.fromNodeId : newNodeId;
      const toId = menuData.anchor === 'right' ? newNodeId : menuData.fromNodeId;
      const newEdge = { id: generateId(), from: fromId, to: toId, label: 'ref' };

      updateDocLocal(d => ({ 
          ...d, 
          nodes: [...d.nodes, newNode],
          edges: [...d.edges, newEdge]
      }));
      setSelection([newNodeId]);
      setEdgeMenu(null);
  };

  const handleDownloadMedia = (e, src, type) => {
      e.stopPropagation();
      const a = document.createElement('a');
      a.href = src;
      a.download = `tapnow_media_${Date.now()}.${type === 'video_media' ? 'mp4' : 'png'}`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
  };

  const handleGenerate = async (genParams) => {
    const generatorNode = panelAnchorNode; 
    const targetType = generatorNode.type; 

    const connectedEdges = docState.edges.filter(e => e.to === generatorNode.id || e.from === generatorNode.id);
    const sourceNodes = connectedEdges.map(e => {
        const otherId = e.from === generatorNode.id ? e.to : e.from;
        return docState.nodes.find(n => n.id === otherId);
    }).filter(Boolean);
    
    const inputUrls = sourceNodes.filter(n => n.type === 'picture').map(n => n.src).filter(Boolean);
    
    let apiMode = genParams.mode;
    if (targetType === 'image') apiMode = inputUrls.length > 0 ? 'img2img' : 'txt2img';
    else if (targetType === 'video') apiMode = inputUrls.length > 0 ? (inputUrls.length > 1 ? 'i2v_first_last' : 'i2v_first') : 'txt2video';
    const finalParams = { ...genParams, mode: apiMode, inputUrls };

    pushHistory(docState);
    updateNodeStrict(generatorNode.id, { _isLoading: true, generation: { ...genParams }, _taskParams: finalParams });

    try {
        let result;
        if (targetType === 'video') {
             result = await ApiClient.generateVideo(finalParams);
             updateNodeStrict(generatorNode.id, { type: 'video_media', src: result.url, _isLoading: false, _taskParams: null, _error: null });
        } else {
             result = await ApiClient.generateImage(finalParams);
             updateNodeStrict(generatorNode.id, { type: 'picture', src: result.urls[0], _isLoading: false, _taskParams: null, _error: null });
        }
    } catch (err) { 
         updateNodeStrict(generatorNode.id, { _isLoading: false, _error: err.message, _taskParams: null }); 
    }
  };

  const activeNodes = useMemo(() => docState.nodes.filter(n => selection.includes(n.id)), [selection, docState.nodes]);
  let panelMode = null; let panelAnchorNode = null; let incomingNodesForPanel = [];
  
  if (activeNodes.length === 1) {
    const selectedNode = activeNodes[0];
    if (selectedNode.type === 'image' || selectedNode.type === 'video') {
        panelMode = selectedNode.type; 
        panelAnchorNode = selectedNode;
        const connectedEdges = docState.edges.filter(e => e.to === panelAnchorNode.id || e.from === panelAnchorNode.id);
        incomingNodesForPanel = connectedEdges.map(e => docState.nodes.find(n => n.id === (e.from === panelAnchorNode.id ? e.to : e.from))).filter(Boolean);
    }
  } 

  return (
    <div className="w-full h-full overflow-hidden relative" onWheel={handleWheel} onPointerDown={handlePointerDown} onDoubleClick={handleDoubleClick}
         onDragOver={(e) => { e.preventDefault(); e.dataTransfer.dropEffect = 'copy'; }}
         onDrop={async (e) => {
             e.preventDefault();
             const rect = containerRef.current.getBoundingClientRect();
             const pos = screenToCanvas(e.clientX - rect.left, e.clientY - rect.top);
             if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
                 const files = Array.from(e.dataTransfer.files).filter(f => f.type.startsWith('image/') || f.type.startsWith('video/'));
                 if (files.length > 0) pushHistory(docState);
                 for (let i = 0; i < files.length; i++) {
                     const file = files[i];
                     try {
                         if (file.type.startsWith('image/')) {
                             const base64 = await fileToBase64(file);
                             const dims = await getImageDimensions(base64); 
                             const size = calculateNodeSize(dims.w, dims.h);
                             createNode('picture', { x: pos.x + i * 30, y: pos.y + i * 30 }, { src: base64, size });
                         } else if (file.type.startsWith('video/')) {
                             // 使用 ObjectURL 防止大视频导致 Firestore 崩溃 (作为本地预览)
                             const url = URL.createObjectURL(file);
                             createNode('video_media', { x: pos.x + i * 30, y: pos.y + i * 30 }, { src: url, size: { w: 360, h: 200 } });
                         }
                     } catch (err) { console.error("拖拽文件失败", err); }
                 }
             }
         }} ref={containerRef}>
      
      <div className="absolute top-4 left-4 right-4 z-50 flex justify-between items-center pointer-events-none">
          <div className="flex gap-2 bg-white/90 backdrop-blur-md p-2 rounded-xl shadow-sm border border-neutral-200 pointer-events-auto">
            <button onClick={onBack} className="p-2 hover:bg-neutral-100 rounded-lg tooltip" title="返回项目列表"><ArrowLeft size={18}/></button>
            <div className="w-px bg-neutral-200 my-1 mx-1"></div>
            <button onClick={() => createNode('text', screenToCanvas(100, 100))} className="p-2 hover:bg-neutral-100 rounded-lg flex items-center gap-2"><Type size={18}/>文本</button>
            <button onClick={() => createNode('image', screenToCanvas(150, 150))} className="p-2 hover:bg-neutral-100 rounded-lg flex items-center gap-2 text-blue-600 font-medium"><Cpu size={18}/>生图节点</button>
            <button onClick={() => createNode('video', screenToCanvas(200, 200))} className="p-2 hover:bg-neutral-100 rounded-lg flex items-center gap-2 text-purple-600 font-medium"><Cpu size={18}/>视频节点</button>
            <div className="w-px bg-neutral-200 my-1 mx-1"></div>
            <button onClick={() => updateDocLocal({ viewport: {x:0, y:0, zoom:1}})} className="p-2 hover:bg-neutral-100 rounded-lg tooltip" title="复位视图"><Maximize size={18}/></button>
          </div>
          <div className="flex gap-2 pointer-events-auto">
            {isSaving && <span className="text-xs text-neutral-400 flex items-center mr-2">正在保存...</span>}
            <button onClick={() => {
                const url = new URL(window.location.href); url.searchParams.set('projectId', projectId);
                try { document.execCommand('copy'); prompt("复制以下链接分享给协作者：", url.toString()); } catch(e) {}
            }} className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-xl flex items-center gap-2 shadow-md shadow-blue-500/20 transition-all font-medium text-sm">
                <Share2 size={16} /> 分享协作
            </button>
          </div>
      </div>

      <div className="absolute inset-0 origin-top-left" style={{ transform: `translate(${docState.viewport.x}px, ${docState.viewport.y}px) scale(${docState.viewport.zoom})`, willChange: 'transform' }}>
        <div className="absolute inset-[-100000px] pointer-events-none z-0" style={{ backgroundImage: `radial-gradient(circle, #cbd5e1 1px, transparent 1px)`, backgroundSize: `${GRID_SIZE}px ${GRID_SIZE}px`, backgroundPosition: `0 0` }} />

        <svg className="absolute inset-0 w-full h-full overflow-visible pointer-events-none z-0">
          {docState.edges.map(edge => {
            const fromNode = docState.nodes.find(n => n.id === edge.from);
            const toNode = docState.nodes.find(n => n.id === edge.to);
            if (!fromNode || !toNode) return null;
            const isSelected = selection.includes(edge.id);
            const x1 = fromNode.position.x + fromNode.size.w; const y1 = fromNode.position.y + fromNode.size.h / 2;
            const x2 = toNode.position.x; const y2 = toNode.position.y + toNode.size.h / 2;
            const ctrlOffset = Math.max(50, Math.abs(x2 - x1) * 0.4);
            const path = `M ${x1} ${y1} C ${x1 + ctrlOffset} ${y1}, ${x2 - ctrlOffset} ${y2}, ${x2} ${y2}`;
            return (
              <g key={edge.id} data-edge-id={edge.id}>
                <path d={path} fill="none" stroke="transparent" strokeWidth={15 / docState.viewport.zoom} className="pointer-events-auto cursor-pointer" onPointerDown={(e) => { e.stopPropagation(); setSelection([edge.id]); }} />
                <path d={path} fill="none" stroke={isSelected ? "#3b82f6" : "#94a3b8"} strokeWidth={(isSelected ? 3 : 2) / docState.viewport.zoom} strokeDasharray={edge.label === 'ref' ? '4 4' : 'none'} className="pointer-events-none" />
              </g>
            );
          })}
          {dragState?.type === 'edge' && (() => {
            const node = docState.nodes.find(n => n.id === dragState.nodeId);
            if (!node) return null;
            let x1, y1, x2, y2;
            if (dragState.anchor === 'right') {
                x1 = node.position.x + node.size.w; y1 = node.position.y + node.size.h / 2;
                x2 = dragState.currentPos.x; y2 = dragState.currentPos.y;
            } else {
                x1 = dragState.currentPos.x; y1 = dragState.currentPos.y;
                x2 = node.position.x; y2 = node.position.y + node.size.h / 2;
            }
            const ctrlOffset = Math.max(50, Math.abs(x2 - x1) * 0.4);
            return <path d={`M ${x1} ${y1} C ${x1 + ctrlOffset} ${y1}, ${x2 - ctrlOffset} ${y2}, ${x2} ${y2}`} fill="none" stroke="#3b82f6" strokeWidth={2 / docState.viewport.zoom} strokeDasharray="4 4" />;
          })()}
          {edgeMenu && (() => {
            const node = docState.nodes.find(n => n.id === edgeMenu.fromNodeId);
            if (!node) return null;
            let x1, y1, x2, y2;
            if (edgeMenu.anchor === 'right') {
                x1 = node.position.x + node.size.w; y1 = node.position.y + node.size.h / 2;
                x2 = edgeMenu.cx; y2 = edgeMenu.cy;
            } else {
                x1 = edgeMenu.cx; y1 = edgeMenu.cy;
                x2 = node.position.x; y2 = node.position.y + node.size.h / 2;
            }
            const ctrlOffset = Math.max(50, Math.abs(x2 - x1) * 0.4);
            return <path d={`M ${x1} ${y1} C ${x1 + ctrlOffset} ${y1}, ${x2 - ctrlOffset} ${y2}, ${x2} ${y2}`} fill="none" stroke="#3b82f6" strokeWidth={2 / docState.viewport.zoom} strokeDasharray="4 4" />;
          })()}
        </svg>

        {docState.nodes.map(node => {
          const isSelected = selection.includes(node.id);
          const isDrawing = drawingNodeId === node.id;
          return (
            <div key={node.id} data-node-id={node.id} onPointerDown={(e) => handleNodePointerDown(e, node.id)}
                 className={`absolute bg-white rounded-2xl shadow-sm border-2 transition-shadow group ${isDrawing ? 'z-50' : 'z-10'} ${isSelected ? 'border-blue-500 shadow-lg shadow-blue-500/20' : 'border-transparent hover:border-neutral-300'}`}
                 style={{ left: node.position.x, top: node.position.y, width: node.size.w, height: node.size.h, cursor: dragState?.type === 'node' && isSelected ? 'grabbing' : 'grab' }}>
              
              <div className="absolute left-[-14px] top-1/2 -translate-y-1/2 w-7 h-7 flex items-center justify-center cursor-crosshair z-20 opacity-0 group-hover:opacity-100 transition-opacity"
                   onPointerDown={(e) => { e.stopPropagation(); setDragState({ type: 'edge', nodeId: node.id, anchor: 'left', currentPos: screenToCanvas(e.clientX - containerRef.current.getBoundingClientRect().left, e.clientY - containerRef.current.getBoundingClientRect().top) }); }}>
                   <div className="w-3.5 h-3.5 bg-blue-500 rounded-full border-[2.5px] border-white shadow-sm pointer-events-none hover:scale-125 transition-transform" />
              </div>
              <div className="absolute right-[-14px] top-1/2 -translate-y-1/2 w-7 h-7 flex items-center justify-center cursor-crosshair z-20 opacity-0 group-hover:opacity-100 transition-opacity"
                   onPointerDown={(e) => { e.stopPropagation(); setDragState({ type: 'edge', nodeId: node.id, anchor: 'right', currentPos: screenToCanvas(e.clientX - containerRef.current.getBoundingClientRect().left, e.clientY - containerRef.current.getBoundingClientRect().top) }); }}>
                   <div className="w-3.5 h-3.5 bg-blue-500 rounded-full border-[2.5px] border-white shadow-sm pointer-events-none hover:scale-125 transition-transform" />
              </div>

              <div className="h-6 bg-neutral-50 border-b border-neutral-100 flex items-center px-3 justify-between rounded-t-xl overflow-hidden">
                 <span className="text-[10px] uppercase font-bold text-neutral-400 tracking-wider flex items-center gap-1">
                   {node.type === 'text' && <><Type size={10}/> 文本</>}
                   {node.type === 'picture' && <><ImageIcon size={10}/> 图片</>}
                   {node.type === 'image' && <><Cpu size={10}/> 生图节点</>}
                   {node.type === 'video' && <><Cpu size={10}/> 视频节点</>}
                   {node.type === 'video_media' && <><Video size={10}/> 视频结果</>}
                 </span>
              </div>

              <div className={`p-3 h-[calc(100%-24px)] w-full flex flex-col relative rounded-b-xl ${isDrawing ? 'overflow-visible' : 'overflow-hidden'}`}>
                {node.type === 'text' && (
                  <textarea className="w-full h-full resize-none outline-none text-sm bg-transparent" placeholder="输入文案或 Prompt..."
                    value={node.content} onChange={(e) => updateDocLocal(d => ({...d, nodes: d.nodes.map(n => n.id === node.id ? {...n, content: e.target.value} : n)}))}
                    onPointerDown={e => e.stopPropagation()} />
                )}
                
                {node.type === 'picture' && (
                  <div className={`w-full h-full flex items-center justify-center bg-neutral-100 rounded-lg relative group ${isDrawing ? 'overflow-visible' : 'overflow-hidden'}`}>
                    {isDrawing ? (
                        <DoodleCanvas src={node.src} zoom={docState.viewport.zoom}
                            onSave={(dataUrl) => { pushHistory(docState); updateNodeStrict(node.id, { src: dataUrl }); setDrawingNodeId(null); }}
                            onCancel={() => setDrawingNodeId(null)} 
                        />
                    ) : node.src ? (
                       <>
                         <img src={node.src} alt="" draggable={false} className="w-full h-full object-contain" />
                         
                         <div className="absolute top-2 right-2 flex gap-1.5 opacity-0 group-hover:opacity-100 transition-opacity z-10">
                             {node.generation?.prompt && (
                                 <button onClick={(e) => { e.stopPropagation(); setVisiblePrompts(p => ({...p, [node.id]: !p[node.id]})); }} className="p-1.5 bg-black/50 hover:bg-black/70 text-white rounded-md" title="显示参数">
                                     <MessageSquare size={14} />
                                 </button>
                             )}
                             <button onClick={(e) => handleDownloadMedia(e, node.src, node.type)} className="p-1.5 bg-black/50 hover:bg-black/70 text-white rounded-md" title="下载图片">
                                 <Download size={14} />
                             </button>
                             <button onClick={(e) => { e.stopPropagation(); setDrawingNodeId(node.id); }} className="p-1.5 bg-black/50 hover:bg-black/70 text-white rounded-md" title="涂鸦修改"><PenTool size={14} /></button>
                         </div>

                         {node.generation?.prompt && visiblePrompts[node.id] && (
                             <div className="absolute bottom-0 left-0 right-0 bg-black/70 backdrop-blur-md text-white/90 p-3 text-xs text-left max-h-36 overflow-y-auto pointer-events-auto select-text cursor-text border-t border-white/10 z-20">
                                 <div className="flex justify-between items-center mb-1 pb-1 border-b border-white/20">
                                     <span className="font-bold text-white/60">生成参数</span>
                                     <button onClick={() => setVisiblePrompts(p => ({...p, [node.id]: false}))} className="hover:text-red-400"><X size={12}/></button>
                                 </div>
                                 <div className="leading-relaxed mb-2">{node.generation.prompt}</div>
                                 <div className="flex flex-wrap gap-2 text-[10px] text-white/60">
                                     {node.generation.aspectRatio && <span className="bg-white/10 px-1.5 py-0.5 rounded">比例: {node.generation.aspectRatio}</span>}
                                     <span className="bg-white/10 px-1.5 py-0.5 rounded">模式: {node.generation.mode}</span>
                                 </div>
                             </div>
                         )}
                       </>
                    ) : <ImageIcon className="text-neutral-300" size={32} />}
                  </div>
                )}

                {node.type === 'image' && (
                   <div className="w-full h-full flex items-center justify-center bg-blue-50/50 rounded-lg border-2 border-dashed border-blue-200 relative">
                     {node._isLoading ? (
                         <div className="flex flex-col items-center gap-2 text-blue-500"><Loader2 className="animate-spin" size={24} /><span className="text-xs font-medium">云端生成中...</span></div>
                     ) : node._error ? (
                         <div className="text-red-500 text-xs text-center p-2">{node._error}<button onClick={() => resumeTask(node)} className="block mx-auto mt-2 px-3 py-1 bg-red-100 rounded-md">重试</button></div>
                     ) : (
                         <div className="flex flex-col items-center justify-center text-center p-4">
                             <ImagePlus className="text-blue-300 mb-2" size={36}/>
                             <span className="text-sm font-bold text-blue-600">图像生成节点</span>
                             <span className="text-[10px] text-neutral-400 mt-2">1. 选中调出输入面板<br/>2. 可连接 <ImageIcon size={10} className="inline"/>图片 作为源参数</span>
                         </div>
                     )}
                   </div>
                )}

                {node.type === 'video' && (
                   <div className="w-full h-full flex items-center justify-center bg-purple-50/50 rounded-lg border-2 border-dashed border-purple-200 relative">
                     {node._isLoading ? (
                         <div className="flex flex-col items-center gap-2 text-purple-500 text-center"><Loader2 className="animate-spin" size={24} /><span className="text-xs font-medium">视频渲染中...</span></div>
                     ) : node._error ? (
                         <div className="text-red-500 text-xs text-center p-2">{node._error}<button onClick={() => resumeTask(node)} className="block mx-auto mt-2 px-3 py-1 bg-red-100 rounded-md">重试</button></div>
                     ) : (
                         <div className="flex flex-col items-center justify-center text-center p-4">
                             <Video className="text-purple-300 mb-2" size={36}/>
                             <span className="text-sm font-bold text-purple-600">视频生成节点</span>

                                  )}

                              )}
                            </>
                        )}
                   </div>
                )}

            </div>
          );
        })}
      </div>

      {dragState?.type === 'box' && selectionBox && <div className="absolute bg-blue-500/10 border border-blue-500 pointer-events-none z-50" style={{ left: selectionBox.x, top: selectionBox.y, width: selectionBox.w, height: selectionBox.h }} />}
      {contextMenu && (

             <button className="w-full text-left px-4 py-2 hover:bg-neutral-100 flex items-center gap-2 text-sm text-neutral-700" onClick={() => { createNode('text', { x: contextMenu.cx, y: contextMenu.cy }); setContextMenu(null); }}><Type size={16}/> 文本</button>

      {edgeMenu && (() => {
          const srcNode = docState.nodes.find(n => n.id === edgeMenu.fromNodeId);
          const canSpawnGens = srcNode && ['text', 'picture'].includes(srcNode.type);

                   <button className="w-full text-left px-4 py-3 hover:bg-neutral-50 flex items-center gap-3 text-sm transition-colors border-b border-neutral-50" 
                           onClick={() => handleCreateFromEdge('text', edgeMenu)}>
                       <div className="bg-neutral-100 text-neutral-600 p-2 rounded-lg"><Type size={18}/></div>

                           <div className="font-bold text-neutral-700">文本节点</div>
                           <div className="text-[10px] text-neutral-400 mt-0.5">添加关联注释或描述</div>
                       </div>
                   </button>
                   
                   {/* 只有图片或文本可以生成新的图片/视频 */}
                   {canSpawnGens && (
                       <>
                           <button className="w-full text-left px-4 py-3 hover:bg-neutral-50 flex items-center gap-3 text-sm transition-colors border-b border-neutral-50" 
                                   onClick={() => handleCreateFromEdge('image', edgeMenu)}>
                               <div className="bg-blue-50 text-blue-600 p-2 rounded-lg"><Cpu size={18}/></div>
                               <div>
                                   <div className="font-bold text-neutral-700">图像生成节点</div>
                                   <div className="text-[10px] text-neutral-400 mt-0.5">将此节点作为参考源</div>
                               </div>
                           </button>
                           
                           <button className="w-full text-left px-4 py-3 hover:bg-neutral-50 flex items-center gap-3 text-sm transition-colors" 
                                   onClick={() => handleCreateFromEdge('video', edgeMenu)}>
                               <div className="bg-purple-50 text-purple-600 p-2 rounded-lg"><Video size={18}/></div>
                               <div>
                                   <div className="font-bold text-neutral-700">视频生成节点</div>
                                   <div className="text-[10px] text-neutral-400 mt-0.5">宣传视频、动画、电影</div>
                               </div>
                           </button>
                       </>
                   )}
              </div>
          );
      })()}

      {panelMode && panelAnchorNode && !drawingNodeId && (
        <GenerationPanel mode={panelMode} anchorNode={panelAnchorNode} incomingNodes={incomingNodesForPanel} viewport={docState.viewport} onGenerate={handleGenerate} />
      )}
    </div>
  );
}

// --- 底部面板组件 ---
function GenerationPanel({ mode, anchorNode, incomingNodes, viewport, onGenerate }) {
    const [prompt, setPrompt] = useState('');
    const [aspectRatio, setAspectRatio] = useState('1:1');
    const [duration, setDuration] = useState('5s');
    const [resolution, setResolution] = useState('720p');
    const [videoModel, setVideoModel] = useState('doubao-seedance-1-5-pro-251215');
    
    const panelRef = useRef(null);
    const [panelStyle, setPanelStyle] = useState({});

    useEffect(() => { setPrompt(anchorNode.generation?.prompt || ''); }, [anchorNode.id, anchorNode.generation?.prompt]);

    useEffect(() => {
        if (!anchorNode || !panelRef.current) return;
        const updatePosition = () => {
            const nodeBottomScreenY = (anchorNode.position.y + anchorNode.size.h) * viewport.zoom + viewport.y;
            const nodeCenterScreenX = (anchorNode.position.x + anchorNode.size.w / 2) * viewport.zoom + viewport.x;
            const panelWidth = panelRef.current.offsetWidth || 320; const panelHeight = panelRef.current.offsetHeight || 120;
            let top = nodeBottomScreenY + 12; let left = nodeCenterScreenX - panelWidth / 2;
            if (top + panelHeight > window.innerHeight - 20) top = (anchorNode.position.y * viewport.zoom + viewport.y) - panelHeight - 12;
            if (left < 20) left = 20; if (left + panelWidth > window.innerWidth - 20) left = window.innerWidth - panelWidth - 20;
            setPanelStyle({ transform: `translate(${left}px, ${top}px)`, opacity: 1 });
        };
        updatePosition();
    }, [anchorNode.position, anchorNode.size, viewport, incomingNodes.length]);

    const handleSend = () => { 
        if (!prompt && mode !== 'video') return; 
        onGenerate({ prompt, aspectRatio, duration, resolution, model: mode === 'video' ? videoModel : DEFAULT_IMAGE_MODEL }); 
    };

    const inputPictures = incomingNodes.filter(n => n.type === 'picture');
    const hasImagesLinked = inputPictures.length;

    return (
        <div ref={panelRef} className="absolute top-0 left-0 w-[340px] bg-white rounded-2xl shadow-2xl border border-neutral-200/60 flex flex-col z-50 transition-transform duration-75 ease-out" style={{ ...panelStyle, opacity: panelStyle.opacity || 0 }} onPointerDown={e => e.stopPropagation()}>
            {mode === 'image' && hasImagesLinked > 0 && (<div className="px-4 py-2 bg-blue-50 border-b border-blue-100 rounded-t-2xl flex items-center gap-2 text-xs text-blue-700"><LinkIcon size={14} /> <span>已读取 {hasImagesLinked} 张图片作为生成源</span></div>)}
            {mode === 'video' && (<div className="px-4 py-2 bg-purple-50 border-b border-purple-100 rounded-t-2xl flex items-center gap-2 text-xs text-purple-700"><Video size={14} /> <span>视频生成 ({hasImagesLinked === 0 ? '文生视频' : (hasImagesLinked === 1 ? '首帧视频' : '首尾帧视频')})</span></div>)}
            
            {hasImagesLinked > 0 && (
                <div className="px-3 pt-2 pb-1 bg-neutral-50/80 border-b border-neutral-100 flex gap-2 overflow-x-auto">
                    {inputPictures.map((pic, i) => (
                        <div key={pic.id} className="relative w-10 h-10 rounded-md border border-neutral-200 overflow-hidden shrink-0 shadow-sm bg-white">
                            <img src={pic.src} className="w-full h-full object-cover" />
                            <div className="absolute bottom-0 right-0 bg-black/60 text-white text-[8px] px-1 rounded-tl">
                                {mode === 'video' ? (i === 0 ? '首帧' : '尾帧') : `图${i+1}`}
                            </div>
                        </div>
                    ))}
                </div>
            )}

            <div className="p-3 relative">
                <textarea className="w-full h-[60px] resize-none outline-none text-sm placeholder-neutral-400 bg-transparent" placeholder="描述任何你想要生成的内容..." value={prompt} onChange={e => setPrompt(e.target.value)} onKeyDown={e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend(); } }} />
                <button onClick={handleSend} disabled={anchorNode._isLoading} className="absolute right-3 bottom-3 w-8 h-8 rounded-full bg-black text-white flex items-center justify-center hover:bg-neutral-800 disabled:opacity-50 transition-all">{anchorNode._isLoading ? <Loader2 size={16} className="animate-spin" /> : <Send size={14} className="-ml-0.5 mt-0.5" />}</button>
            </div>

            <div className="px-3 pb-3 flex flex-wrap gap-2 text-xs border-t border-neutral-100 pt-2">
                <div className="flex items-center gap-1 bg-neutral-100 text-neutral-600 px-2 py-1 rounded-md">
                    <Settings2 size={12}/> 
                    {mode === 'video' ? (
                        <select value={videoModel} onChange={e => setVideoModel(e.target.value)} className="bg-transparent outline-none cursor-pointer">
                            <option value="doubao-seedance-1-5-pro-251215">即梦1.5 Pro</option>
                            <option value="Hailuo">海螺 MiniMax</option>
                        </select>
                    ) : (
                        'Nano Banana'
                    )}
                </div>
                {mode !== 'video' ? (
                    <select value={aspectRatio} onChange={e => setAspectRatio(e.target.value)} className="bg-neutral-100 text-neutral-600 px-2 py-1 rounded-md outline-none cursor-pointer hover:bg-neutral-200 appearance-none">
                        <option value="1:1">1:1 方形</option><option value="16:9">16:9 横屏</option><option value="9:16">9:16 竖屏</option>
                    </select>
                ) : (
                    <>
                        <select value={duration} onChange={e => setDuration(e.target.value)} className="bg-neutral-100 text-neutral-600 px-2 py-1 rounded-md outline-none cursor-pointer hover:bg-neutral-200 appearance-none">
                            <option value="5s">5秒</option>
                            <option value="10s">10秒</option>
                        </select>
                        <select value={resolution} onChange={e => setResolution(e.target.value)} className="bg-neutral-100 text-neutral-600 px-2 py-1 rounded-md outline-none cursor-pointer hover:bg-neutral-200 appearance-none">
                            <option value="720p">720p</option>
                            <option value="1080p">1080p</option>
                            <option value="2K">2K</option>
                            <option value="4K">4K</option>
                        </select>
                    </>
                )}
            </div>
        </div>
    );
}

// --- 涂鸦画布组件 ---
function DoodleCanvas({ src, onSave, onCancel, zoom }) {
    const containerRef = useRef(null);
    const canvasRef = useRef(null);
    const [isDrawing, setIsDrawing] = useState(false);
    const [color, setColor] = useState('#ff0000');
    const ctxRef = useRef(null);

    useEffect(() => {
        if (!containerRef.current || !canvasRef.current) return;
        const width = containerRef.current.offsetWidth;
        const height = containerRef.current.offsetHeight;
        const canvas = canvasRef.current;
        canvas.width = width;
        canvas.height = height;
        
        const ctx = canvas.getContext('2d');
        ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.lineWidth = 5;
        ctxRef.current = ctx;

        const loadAndRender = async () => {
            let finalSrc = src;
            if (src && !src.startsWith('data:')) {
                try {
                    const res = await fetch(src, { mode: 'cors' });
                    const blob = await res.blob();
                    finalSrc = await fileToBase64(blob);
                } catch (e) {
                    try {
                        const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(src)}`;
                        const res = await fetch(proxyUrl);
                        const blob = await res.blob();
                        finalSrc = await fileToBase64(blob);
                    } catch (err) {
                        console.warn("涂鸦底图渲染失败");
                        ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, width, height);
                        return;
                    }
                }
            }

            if (finalSrc) {
                const img = new Image();
                img.crossOrigin = 'anonymous'; 
                img.onload = () => {
                    const scale = Math.min(width / img.width, height / img.height);
                    const x = (width / 2) - (img.width / 2) * scale;
                    const y = (height / 2) - (img.height / 2) * scale;
                    ctx.drawImage(img, x, y, img.width * scale, img.height * scale);
                };
                img.src = finalSrc;
            } else {
                ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, width, height);
            }
        };

        loadAndRender();
    }, [src]);

    useEffect(() => {
        if (ctxRef.current) ctxRef.current.strokeStyle = color;
    }, [color]);

    const getPos = (e) => {
        const rect = canvasRef.current.getBoundingClientRect();
        return { x: (e.clientX - rect.left) / zoom, y: (e.clientY - rect.top) / zoom };
    };

    const startDrawing = (e) => { 
        e.stopPropagation(); 
        setIsDrawing(true); 
        const { x, y } = getPos(e); 
        ctxRef.current.beginPath(); 
        ctxRef.current.moveTo(x, y); 
        ctxRef.current.lineTo(x, y); 
        ctxRef.current.stroke();
    };
    const draw = (e) => { e.stopPropagation(); if (!isDrawing) return; const { x, y } = getPos(e); ctxRef.current.lineTo(x, y); ctxRef.current.stroke(); };
    const stopDrawing = (e) => { e.stopPropagation(); setIsDrawing(false); ctxRef.current.closePath(); };

    const handleSave = () => {
        try {
            const dataUrl = canvasRef.current.toDataURL('image/jpeg', 0.80);
            onSave(dataUrl);
        } catch (e) {
            console.error("涂鸦保存失败:", e);
            alert("保存涂鸦失败：可能由于原图安全限制无法提取画布，请重试。");
        }
    };

    return (
        <div ref={containerRef} className="absolute inset-0 z-30 bg-neutral-100" onPointerDown={e => e.stopPropagation()}>
            <canvas
                ref={canvasRef}
                style={{ width: '100%', height: '100%', touchAction: 'none', cursor: 'crosshair' }}
                onPointerDown={startDrawing} onPointerMove={draw} onPointerUp={stopDrawing} onPointerLeave={stopDrawing}
            />
            
            <div className="absolute -bottom-16 left-1/2 -translate-x-1/2 flex items-center gap-3 bg-white p-2 px-4 rounded-full shadow-xl border border-neutral-200 z-50 w-max cursor-default" onPointerDown={e => e.stopPropagation()}>
                <div className="flex gap-2 border-r border-neutral-200 pr-3">
                    {['#ef4444', '#000000', '#ffffff', '#3b82f6', '#22c55e'].map(c => (
                        <button key={c} onPointerDown={(e) => { e.stopPropagation(); setColor(c); }} 
                                className={`w-6 h-6 rounded-full border-[2px] ${color === c ? 'border-neutral-400 scale-110' : 'border-neutral-200'} shadow-sm transition-transform`} 
                                style={{ backgroundColor: c }} />
                    ))}
                </div>
                <div className="flex gap-2 pl-1">
                    <button onPointerDown={(e) => { e.stopPropagation(); onCancel(); }} className="p-2 bg-neutral-100 rounded-full text-neutral-600 hover:bg-neutral-200 hover:text-red-500 transition-all">
                        <X size={16}/>
                    </button>
                    <button onPointerDown={(e) => { e.stopPropagation(); handleSave(); }} className="p-2 bg-blue-500 rounded-full text-white hover:bg-blue-600 transition-all">
                        <Check size={16}/>
                    </button>
                </div>
            </div>
        </div>
    );
}
